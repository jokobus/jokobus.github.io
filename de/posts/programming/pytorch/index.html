PyTorch hat sich als führendes Deep-Learning-Framework in Forschung und Industrie etabliert [paszke2019] . Es zeichnet sich durch dynamische Berechnungsgraphen (Eager Execution), unkomplizierte GPU-Beschleunigung und eine intuitive, pythonische Syntax aus.

PyTorch vs. TensorFlow vs. Scikit-Learn

Die Wahl des richtigen Werkzeugs richtet sich nach dem Einsatzzweck:

FrameworkStärkenTypischer Einsatzbereich
Scikit-LearnKlassische Algorithmen, schnelles Prototyping, flaches MLTabellendaten, Regression, Random Forests, SVMs, Clustering
PyTorchDynamische Graphen (Eager Execution), Spitzenforschung, flexibles DebuggingDeep Learning, Computer Vision, Transformers / LLMs, GNNs
TensorFlow / KerasStatische Graphenkompilierung, industrieller ProduktionsbetriebMobile Geräte (TFLite) und hochskalierende Serverinfrastruktur

1. Was ist ein Tensor?

In der Mathematik bezeichnet ein Tensor ein multilineares algebraisches Gebilde über Vektorräumen. Im Kontext von Deep Learning gilt pragmatisch:

  • Ein 0-D Tensor ist ein Skalar (z. B. $s = 42$).
  • Ein 1-D Tensor ist ein Vektor (z. B. $v = [1, 2, 3]$).
  • Ein 2-D Tensor ist eine Matrix aus Zeilen und Spalten (z. B. ein Graustufenbild oder eine Gewichtsmatrix).
  • Ein 3-D+ Tensor ist ein mehrdimensionales Datenfeld (z. B. Batch-Größe $\times$ Farbkanäle $\times$ Höhe $\times$ Breite für Bilder und Videos).

Praktische Tensor-Operationen

Das folgende Skript veranschaulicht Erstellung, Datentypen, Formen, Hardwarezuweisung und NumPy-Interoperabilität:

📄 Tensors.py - Grundlegende Tensor-Operationen
python
import torch
import numpy as np

# Initialize a Tensor
## Directly from data
data = [[1,2], [3,4]]
x_data = torch.tensor(data)
print(f"From Data: \n{x_data} \n")
## From another NumPy Array
np_array = np.array(data)
x_np = torch.from_numpy(np_array)
print(f"From NumPy: \n{x_np} \n")
## From another tensor
x_ones = torch.ones_like(x_data)
print(f"Ones Tensor: \n{x_ones} \n")
x_rand = torch.rand_like(x_data, dtype=torch.float)
print(f"Random Tensor: \n{x_rand} \n")
## Create tensor with specific shape
shape = (2,3,)
rand_tensor = torch.rand(shape)
ones_tensor = torch.ones(shape)
zeros_tensor = torch.zeros(shape)
print(f"Random 2x3 Tensor: \n{rand_tensor} \n")
print(f"Ones 2x3 Tensor: \n{ones_tensor} \n")
print(f"Zeros 2x3 Tensor: \n{zeros_tensor}")
## Print attributes of the tensor
print(f"Shape of Random 2x3 Tensor: \n{rand_tensor.shape} \n")
print(f"Datatype of Random 2x3 Tensor: \n{rand_tensor.dtype} \n")
print(f"Device Random 2x3 Tensor is stored on: \n{rand_tensor.device}")

# Move the current tensor to an accelerator
# Wait for later torch versions
# if torch.accelerator.is_available():
#    tesor = rand_tensor.to(torch.accelerator.current_accelerator())
# For now stick to
device = 'cuda' if torch.cuda.is_available() else 'cpu'
rand_tensor.to(device)

2. Vollständiges Klassifikationsbeispiel: FashionMNIST

Um das Zusammenspiel von Datensätzen, Modellen, Verlustfunktionen und Optimierern zu verstehen, trainieren wir ein mehrschichtiges Perzeptron auf dem FashionMNIST-Datensatz:

  1. Datensatz & DataLoader: Automatisches Herunterladen und Batchen von Kleidungsbildern ($28 \times 28$ Graustufen).
  2. Modellarchitektur: Definition eines 3-schichtigen Netzes über nn.Module.
  3. Verlustfunktion & Optimierer: Kreuzentropie (nn.CrossEntropyLoss) und stochastischer Gradientenabstieg (torch.optim.SGD).
  4. Trainings- und Evaluierungsschleife: Vorwärtspass, Gradientenberechnung via .backward() und Parameteranpassung.

Vollständiges Trainingsskript

📄 FashionMNIST.py - Vollständige Trainingspipeline
python
import torch
from torch import nn
from torch.utils.data import DataLoader
from torchvision import datasets
from torchvision.transforms import ToTensor

# 1. Get data
def getData():
    # Download training data from open datasets.
    train_data = datasets.FashionMNIST(
        root="data",
        train=True,
        download=True,
        transform=ToTensor(),
    )

    # Download test data from open datasets.
    test_data = datasets.FashionMNIST(
        root="data",
        train=False,
        download=True,
        transform=ToTensor(),
    )
    return train_data, test_data

def printDataLoader(dataloader):
    for x, y in dataloader: 
        print(f"Shape of x [N, C, H, W]: {x.shape}")
        print(f"Shape of y: {y.shape} {y.type}")
        break

# 2. Define model
class NeuralNetwork(nn.Module):
    def __init__(self):
        super().__init__()
        self.flatten = nn.Flatten()
        self.linear_relu_stack = nn.Sequential(
            nn.Linear(28*28, 512),  # first layer, sample down from 784 to 512
            nn.ReLU(),              # activation function
            nn.Linear(512, 512),    # second layer
            nn.ReLU(),              # second activation function
            nn.Linear(512,10)       # third layer to output
        )       
    def forward(self,x):
        x = self.flatten(x)
        logits = self.linear_relu_stack(x)
        return logits

# 3. Define the training function
def train(dataloader, model, loss_fn, optimizer, device="cpu"):
    size = len(dataloader.dataset)
    model.train()
    for batch, (x,y) in enumerate(dataloader):
        x,y = x.to(device), y.to(device)
        # Compute prediction error
        pred = model(x)
        loss = loss_fn(pred,y)
        # Backpropagation
        loss.backward()
        optimizer.step()
        optimizer.zero_grad()

        if batch % 100==0:
            loss, current = loss.item(), (batch + 1) * len(x)
            print(f"loss: {loss:>7f} [{current:>5d}/{size:>5d}]")

# 4. Define the testing function
def test(dataloader, model, loss_fn, device="cpu"):
    size = len(dataloader.dataset)
    num_batches = len(dataloader)
    model.eval()
    test_loss, correct = 0,0
    with torch.no_grad():
        for x,y in dataloader:
            x,y = x.to(device), y.to(device)
            pred = model(x)
            test_loss += loss_fn(pred, y).item()
            correct += (pred.argmax(1) == y).type(torch.float).sum().item()
    test_loss /= num_batches
    correct /= size
    print(f"Test Error: \n Accuracy: {(100*correct):>0.1f}%, Avg loss: {test_loss:>8f} \n")


if __name__ == "__main__":
    # Get the data
    batch_size = 64
    train_data, test_data = getData()
    # Create data loaders
    train_dataloader = DataLoader(train_data, batch_size=batch_size)
    test_dataloader = DataLoader(test_data, batch_size=batch_size)

    # Determine accelerator
    device = "cuda" if torch.cuda.is_available() else "cpu"
    print(f"Using {device} device")
    
    # Initialize model, loss function and optimizer
    model = NeuralNetwork().to(device)
    loss_fn = nn.CrossEntropyLoss()
    optimizer = torch.optim.SGD(model.parameters(), lr=1e-3)
    print(model)

    # Train the model
    epochs = 5
    for t in range(epochs):
        print(f"Epoch {t+1}\n-------------------------------")
        train(train_dataloader, model, loss_fn, optimizer, device)
        test(test_dataloader, model, loss_fn, device)
    
    # Save the model
    torch.save(model.state_dict(), "model.pth")
    print("Saved PyTorch Model State to model.pth")

    # Load the model
    model = NeuralNetwork().to(device)
    model.load_state_dict(torch.load("model.pth", weights_only=True))

    classes = [
         "T-shirt/top",
        "Trouser",
        "Pullover",
        "Dress",
        "Coat",
        "Sandal",
        "Shirt",
        "Sneaker",
        "Bag",
        "Ankle boot",
    ]

    model.eval()
    x,y = test_data[0][0], test_data[0][1]
    with torch.no_grad():
        x=x.to(device)
        pred = model(x)
        predicted, actual = classes[pred[0].argmax(0)], classes[y]
        print(f'Predicted: "{predicted}", Actual: "{actual}"')
    print("Done!")

Kernpunkte

  • Dynamischer Berechnungsgraph: PyTorch erzeugt den Graphen zur Laufzeit (autograd), was reguläre Python-Kontrollstrukturen (Schleifen, if-Verzweigungen) in Modellschichten erlaubt.
  • Hardware-Zuweisung: Das Verschieben von Tensoren und Modellen zwischen Hauptspeicher (CPU) und Grafikkarte (GPU) geschieht nahtlos mit .to(device).
  • Modularer Aufbau: Die Trennung in Dataset, DataLoader, nn.Module und optim erleichtert die Entwicklung und Skalierung komplexer KI-Architekturen erheblich.
